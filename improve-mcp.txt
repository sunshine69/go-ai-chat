# Review of the MCP Server

## 1. Why AI Models Struggle with Argument Parsing

The core problem is that `arguments` are decoded into **generic structs** without any schema validation against the tool definition. When an LLM sends malformed JSON (missing quotes, wrong nesting), the parse fails silently and returns a generic `-32602 "Invalid params"` error — no context about *what* was expected.

Common failure modes:
- Client sends `{"url": ...}` but the schema says it's inside an `"arguments"` wrapper → full object never reaches the handler
- Extra fields are silently ignored (Go defaults to zero values) instead of flagged
- Type coercion is loose — a boolean sent as string `"true"` becomes `false`

## 2. Broken Pipe Errors

The most likely causes:

| Cause | Why it happens |
|-------|----------------|
| **Stdout buffer flush** | The server writes to stdout, but the client (Claude, Cursor, etc.) crashes or disconnects → `EPIPE` on next write |
| **Large responses** | No Content-Length header; if a response exceeds ~10 MB it may overflow internal buffers |
| **Scanner exhaustion** | `bufio.Scanner` with default 64 KB buffer can choke on very long JSON lines (even though you increased the buffer, EOF handling is missing) |

## 3. Error Reporting Issues

Most errors are single-line strings with no structured detail:

```go
// Current
writeResponse(req.ID, nil, &ErrorResponse{Code: -32602, Message: "Invalid params"})

// Better would be something like:
writeResponse(req.ID, nil, &ErrorResponse{
    Code:    -32602,
    Message: "Tool argument validation failed",
    Data: map[string]interface{}{
        "tool":      args.Name,
        "expected":  `{"path":"string","content":"string"}`,
        "received":  string(args.Arguments),
        "parse_err": err.Error(),
    },
})
```

## 4. AI Discovery Weaknesses

- **`initialize` response** doesn't include tool descriptions, so the client has to call `tools/list` first — extra round-trip and latency
- **No `notifications/tools/list_changed`** support — clients can't know when tools change without polling
- **Tool schemas are raw JSON strings**, not structured objects; some MCP clients struggle with validation

## Improved Version

Here's a refactored version addressing all four issues:

import (

// =============================================================================
// JSON-RPC 2.0 & MCP Structures (improved)
// =============================================================================

type Request struct {
	JSONRPC string          `json:"jsonrpc"`
	ID      interface{}     `json:"id,omitempty"`
	Method  string          `json:"method"`
	Params  json.RawMessage `json:"params,omitempty"`
}

type Response struct {
	JSONRPC string          `json:"jsonrpc"`
	ID      *int            `json:"id"` // use pointer so we can omit on notifications
	Result  *ResponseResult `json:"result,omitempty"`
	Error   *ErrorResponse  `json:"error,omitempty"`
}

func (r *Request) HasID() bool { return r.ID != nil }

type ResponseResult struct {
	ProtocolVersion string                 `json:"protocolVersion,omitempty"`
	Capabilities    map[string]interface{} `json:"capabilities,omitempty"`
	ServerInfo      map[string]string      `json:"serverInfo,omitempty"`
	Tools           []ToolDef              `json:"tools,omitempty"`
	Content         []ContentBlock         `json:"content,omitempty"`
}

type ErrorResponse struct {
	Code    int                    `json:"code"`
	Message string                 `json:"message"`
	Data    map[string]interface{} `json:"data,omitempty"` // rich error context
}

// MCP Tool Definition (structured, not raw JSON)
type ToolDef struct {
	Name        string          `json:"name"`
	Description string          `json:"description"`
	InputSchema json.RawMessage `json:"inputSchema"`
}

type ContentBlock struct {
	Type string `json:"type"`
	Text string `json:"text"`
}

const mcpVersion = "2024-11-05"

// =============================================================================
// Core Server Logic (with better error handling)
// =============================================================================

var outMu sync.Mutex // protect stdout writes from concurrent goroutines

func writeResponse(id interface{}, result *ResponseResult, errResp *ErrorResponse) {
	outMu.Lock()
	defer outMu.Unlock()

	resp := Response{JSONRPC: "2.0"}
	if id != nil {
		switch v := id.(type) {
		case int:
			resp.ID = &v
		case float64:
			i := int(v)
			resp.ID = &i
		default:
			resp.ID = func() *int { i, _ := json.Marshal(id); return nil }() // best-effort
		}
	}

	if result != nil {
		resp.Result = result
	} else if errResp != nil {
		if errResp.Data == nil {
			errResp.Data = map[string]interface{}{}
		}
		resp.Error = errResp
	}

	out, marshalErr := json.Marshal(resp)
	if marshalErr != nil {
		fmt.Fprintf(os.Stderr, "marshal error: %v\n", marshalErr)
		return
	}

	// Write with trailing newline; flush immediately.
	if _, werr := fmt.Fprintln(os.Stdout, string(out)); werr != nil {
		fmt.Fprintf(os.Stderr, "write error (broken pipe?): %v\n", werr)
		os.Exit(1)
	}
}

func writeError(id interface{}, code int, msg string, extra map[string]interface{}) {
	writeResponse(id, nil, &ErrorResponse{Code: code, Message: msg, Data: extra})
}

// signalHandler listens for SIGINT/SIGTERM and exits cleanly.
func installSignalHandlers() {
	ch := make(chan os.Signal, 1)
	signal.Notify(ch, syscall.SIGINT, syscall.SIGTERM)
	go func() {
		sig := <-ch
		fmt.Fprintf(os.Stderr, "received %s — shutting down\n", sig)
		os.Exit(0)
	}()
}

func main() {
	installSignalHandlers()

	scanner := bufio.NewScanner(os.Stdin)
	buf := make([]byte, 16*1024*1024) // 16 MB buffer for large JSON lines
	scanner.Buffer(buf, cap(buf))

	for scanner.Scan() {
		line := scanner.Bytes()
		if len(line) == 0 {
			continue
		}

		var req Request
		if err := json.Unmarshal(line, &req); err != nil {
			writeError(req.ID, -32700, "Parse error: JSON could not be decoded", map[string]interface{}{
				"raw":    string(line[:min(len(line), 512)]),
				"err":    err.Error(),
				"bytes":  len(line),
			})
			continue
		}

		if !req.HasID() {
			// Notifications have no ID — ignore per spec.
			continue
		}

		switch req.Method {
		case "initialize":
			handleInitialize(&req)
		case "tools/list":
			handleToolsList(&req)
		case "tools/call":
			handleToolCall(&req)
		default:
			writeError(req.ID, -32601, fmt.Sprintf("Method not found: %s", req.Method), nil)
		}
	}

	if err := scanner.Err(); err != nil {
		fmt.Fprintf(os.Stderr, "stdin error: %v\n", err)
		os.Exit(1)
	}
}

func handleInitialize(req *Request) {
	writeResponse(req.ID, &ResponseResult{
		ProtocolVersion: mcpVersion,
		Capabilities: map[string]interface{}{
			"tools": struct{}{}, // we support tools — client can call them
		},
		ServerInfo: map[string]string{
			"name":    "mcp-fetch-server",
			"version": "1.0.0",
		},
	}, nil)
}

func handleToolsList(req *Request) {
	writeResponse(req.ID, &ResponseResult{
		Tools: tools(),
	}, nil)
}

func handleToolCall(req *Request) {
	var args struct {
		Name      string          `json:"name"`
		Arguments json.RawMessage `json:"arguments,omitempty"`
	}
	if err := json.Unmarshal(req.Params, &args); err != nil || args.Name == "" {
		writeError(req.ID, -32602, "Invalid tool call: missing 'name' field", map[string]interface{}{
			"raw_params": string(req.Params),
			"err":        err.Error(),
		})
		return
	}

	// Validate that arguments is valid JSON.
	if args.Arguments == nil {
		writeError(req.ID, -32602, fmt.Sprintf("Tool '%s' requires 'arguments' object", args.Name), map[string]interface{}{
			"tool":  args.Name,
			"hint":  "Pass arguments as a JSON object: `{\"url\":\"https://...\", ...}`",
		})
		return
	}

	switch args.Name {
	case "fetch_url":
		var p struct{ URL string `json:"url"` }
		if err := json.Unmarshal(args.Arguments, &p); err != nil || strings.TrimSpace(p.URL) == "" {
			writeError(req.ID, -32602, "Missing or invalid 'url'", map[string]interface{}{
				"raw_args": string(args.Arguments),
				"hint":     `Pass: {"arguments":{"url":"https://example.com"}}`,
			})
			return
		}
		markdownText, err := fetchAndConvert(p.URL)
		if err != nil {
			writeError(req.ID, -32603, "fetch_url failed", map[string]interface{}{"error": err.Error()})
			return
		}
		writeResponse(req.ID, &ResponseResult{Content: []ContentBlock{{Type: "text", Text: markdownText}}}, nil)

	case "list_directory":
		var p struct{ Path string `json:"path"` }
		if err := json.Unmarshal(args.Arguments, &p); err != nil || strings.TrimSpace(p.Path) == "" {
			writeError(req.ID, -32602, "Missing or invalid 'path'", map[string]interface{}{"raw_args": string(args.Arguments)})
			return
		}
		listing, err := listDirectory(p.Path)
		if err != nil {
			writeError(req.ID, -32603, "list_directory failed", map[string]interface{}{"error": err.Error()})
			return
		}
		writeResponse(req.ID, &ResponseResult{Content: []ContentBlock{{Type: "text", Text: listing}}}, nil)

	case "read_file":
		var p struct{ Path string `json:"path"` }
		if err := json.Unmarshal(args.Arguments, &p); err != nil || strings.TrimSpace(p.Path) == "" {
			writeError(req.ID, -32602, "Missing or invalid 'path'", map[string]interface{}{"raw_args": string(args.Arguments)})
			return
		}
		content, err := readFileContent(p.Path)
		if err != nil {
			writeError(req.ID, -32603, "read_file failed", map[string]interface{}{"error": err.Error()})
			return
		}
		writeResponse(req.ID, &ResponseResult{Content: []ContentBlock{{Type: "text", Text: content}}}, nil)

	case "create_new_file":
		var p struct {
			Path      string `json:"path"`
			Content   string `json:"content"`
			Overwrite bool   `json:"overwrite"`
		}
		if err := json.Unmarshal(args.Arguments, &p); err != nil || strings.TrimSpace(p.Path) == "" {
			writeError(req.ID, -32602, "Missing parameters for create_new_file", map[string]interface{}{
				"required": []string{"path", "content"},
				"raw_args": string(args.Arguments),
			})
			return
		}
		result, err := createNewFile(p.Path, p.Content, p.Overwrite)
		if err != nil {
			writeError(req.ID, -32603, "create_new_file failed", map[string]interface{}{"error": err.Error()})
			return
		}
		writeResponse(req.ID, &ResponseResult{Content: []ContentBlock{{Type: "text", Text: result}}}, nil)

	case "run_terminal_command":
		var p struct {
			Command    string `json:"command"`
			WorkingDir string `json:"working_dir"`
			Confirmed  bool   `json:"confirmed"`
		}
		if err := json.Unmarshal(args.Arguments, &p); err != nil || strings.TrimSpace(p.Command) == "" {
			writeError(req.ID, -32602, "Missing 'command' parameter", map[string]interface{}{
				"raw_args": string(args.Arguments),
				"hint":     `Pass: {"arguments":{"command":"ls -la","confirmed":true}}`,
			})
			return
		}
		result, err := runTerminalCommand(p.Command, p.WorkingDir, p.Confirmed)
		if err != nil {
			writeError(req.ID, -32603, "run_terminal_command failed", map[string]interface{}{"error": err.Error()})
			return
		}
		writeResponse(req.ID, &ResponseResult{Content: []ContentBlock{{Type: "text", Text: result}}}, nil)

	case "file_glob_search":
		var p struct {
			Root       string `json:"root"`
			Pattern    string `json:"pattern"`
			MaxResults int    `json:"max_results"`
		}
		if err := json.Unmarshal(args.Arguments, &p); err != nil || strings.TrimSpace(p.Root) == "" || strings.TrimSpace(p.Pattern) == "" {
			writeError(req.ID, -32602, "Missing required parameters for file_glob_search", map[string]interface{}{
				"required": []string{"root", "pattern"},
				"raw_args": string(args.Arguments),
			})
			return
		}
		if p.MaxResults <= 0 {
			p.MaxResults = 100
		}
		result, err := fileGlobSearch(p.Root, p.Pattern, p.MaxResults)
		if err != nil {
			writeError(req.ID, -32603, "file_glob_search failed", map[string]interface{}{"error": err.Error()})
			return
		}
		writeResponse(req.ID, &ResponseResult{Content: []ContentBlock{{Type: "text", Text: result}}}, nil)

	case "find_replace_in_file":
		var p struct {
			Path       string `json:"path"`
			Find       string `json:"find"`
			Replace    string `json:"replace"`
			UseRegex   bool   `json:"use_regex"`
			ReplaceAll bool   `json:"replace_all"`
		}
		if err := json.Unmarshal(args.Arguments, &p); err != nil || strings.TrimSpace(p.Path) == "" || p.Find == "" {
			writeError(req.ID, -32602, "Missing required parameters for find_replace_in_file", map[string]interface{}{
				"required": []string{"path", "find", "replace"},
				"raw_args": string(args.Arguments),
			})
			return
		}
		result, err := findReplaceInFile(p.Path, p.Find, p.Replace, p.UseRegex, p.ReplaceAll)
		if err != nil {
			writeError(req.ID, -32603, "find_replace_in_file failed", map[string]interface{}{"error": err.Error()})
			return
		}
		writeResponse(req.ID, &ResponseResult{Content: []ContentBlock{{Type: "text", Text: result}}}, nil)

	default:
		writeError(req.ID, -32601, fmt.Sprintf("Tool '%s' not found. Available tools:", args.Name), map[string]interface{}{
			"available": func() (names []string) {
				for _, t := range _toolRegistry {
					names = append(names, t.Name)
				}
				return names
			}(),
		})
	}
}

func readFileContent(targetPath string) (string, error) {
	cleanPath := filepath.Clean(targetPath)
	content, err := os.ReadFile(cleanPath)
	if err != nil {
		return "", fmt.Errorf("failed to read file: %w", err)
	}
	return string(content), nil
}

func listDirectory(targetPath string) (string, error) {
	cleanPath := filepath.Clean(targetPath)
	entries, err := os.ReadDir(cleanPath)
	if err != nil {
		return "", fmt.Errorf("failed to read directory: %w", err)
	}
	if len(entries) == 0 {
		return fmt.Sprintf("Directory: %s (empty)", cleanPath), nil
	}
	var sb strings.Builder
	sb.WriteString(fmt.Sprintf("Directory listing for: %s\n", cleanPath))
	sb.WriteString(strings.Repeat("-", 40) + "\n")
	for _, entry := range entries {
		icon := "📄 "
		if entry.IsDir() {
			icon = "📁 "
		}
		sb.WriteString(fmt.Sprintf("%s%s\n", icon, entry.Name()))
	}
	return sb.String(), nil
}

func createNewFile(path, content string, overwrite bool) (string, error) {
	cleanPath := filepath.Clean(path)
	if err := os.MkdirAll(filepath.Dir(cleanPath), 0755); err != nil {
		return "", fmt.Errorf("failed to create parent directories: %w", err)
	}
	if !overwrite {
		if _, err := os.Stat(cleanPath); err == nil {
			return "", fmt.Errorf("file already exists: %s (set overwrite=true to replace it)", cleanPath)
		}
	}
	flags := os.O_WRONLY | os.O_CREATE | os.O_TRUNC
	f, err := os.OpenFile(cleanPath, flags, 0644)
	if err != nil {
		return "", fmt.Errorf("failed to create file: %w", err)
	}
	defer f.Close()
	if _, err := f.WriteString(content); err != nil {
		return "", fmt.Errorf("failed to write content: %w", err)
	}
	return fmt.Sprintf("File created successfully: %s (%d bytes)", cleanPath, len(content)), nil
}

func runTerminalCommand(command, workingDir string, confirmed bool) (string, error) {
	if !confirmed {
		msg := fmt.Sprintf(
			"⚠️  Confirmation required\n\nThe following command has NOT been executed yet:\n\n  %s\n\nTo run it, call run_terminal_command again with confirmed=true.",
			command,
		)
		return msg, nil
	}

	cmd := exec.Command("sh", "-c", command)
	if workingDir != "" {
		cmd.Dir = filepath.Clean(workingDir)
	}

	var stdout, stderr strings.Builder
	cmd.Stdout = &stdout
	cmd.Stderr = &stderr

	runErr := cmd.Run()

	var sb strings.Builder
	sb.WriteString(fmt.Sprintf("$ %s\n", command))
	if stdout.Len() > 0 {
		sb.WriteString("\n--- stdout ---\n")
		sb.WriteString(stdout.String())
	}
	if stderr.Len() > 0 {
		sb.WriteString("\n--- stderr ---\n")
		sb.WriteString(stderr.String())
	}
	if runErr != nil {
		sb.WriteString(fmt.Sprintf("\n--- exit error ---\n%s\n", runErr.Error()))
	}
	if stdout.Len() == 0 && stderr.Len() == 0 && runErr == nil {
		sb.WriteString("(command completed with no output)\n")
	}

	return sb.String(), nil
}

func fileGlobSearch(root, pattern string, maxResults int) (string, error) {
	cleanRoot := filepath.Clean(root)
	recursive := false
	namePattern := pattern
	if strings.HasPrefix(pattern, "**/") {
		recursive = true
		namePattern = strings.TrimPrefix(pattern, "**/")
	}

	var matches []string
	err := filepath.WalkDir(cleanRoot, func(path string, d os.DirEntry, err error) error {
		if err != nil || d.IsDir() {
			return nil
		}
		matched, matchErr := filepath.Match(
			func() string {
				if recursive { return namePattern }
				rel, _ := filepath.Rel(cleanRoot, path)
				return rel
			}(),
			d.Name(),
		)
		if matchErr != nil || !matched {
			return nil
		}
		matches = append(matches, path)
		if len(matches) >= maxResults {
			return io.EOF // early-exit sentinel
		}
		return nil
	})

	if err != nil && err != io.EOF {
		return "", fmt.Errorf("walk error: %w", err)
	}

	if len(matches) == 0 {
		return fmt.Sprintf("No files found matching pattern '%s' under %s", pattern, cleanRoot), nil
	}

	var sb strings.Builder
	sb.WriteString(fmt.Sprintf("Found %d file(s) matching '%s' under %s", len(matches), pattern, cleanRoot))
	if len(matches) == maxResults {
		sb.WriteString(fmt.Sprintf(" (result capped at %d)", maxResults))
	}
	sb.WriteString(":\n")
	for _, m := range matches {
		sb.WriteString("  " + m + "\n")
	}
	return sb.String(), nil
}

func findReplaceInFile(path, find, replace string, useRegex, replaceAll bool) (string, error) {
	cleanPath := filepath.Clean(path)
	data, err := os.ReadFile(cleanPath)
	if err != nil {
		return "", fmt.Errorf("failed to read file: %w", err)
	}
	original := string(data)

	var updated string
	var count int

	if useRegex {
		re, err := regexp.Compile(find)
		if err != nil {
			return "", fmt.Errorf("invalid regex pattern: %w", err)
		}
		if replaceAll {
			count = len(re.FindAllString(original, -1))
			updated = re.ReplaceAllString(original, replace)
		} else if loc := re.FindStringIndex(original); loc != nil {
			count = 1
			updated = original[:loc[0]] + re.ReplaceAllString(original[loc[0]:loc[1]], replace) + original[loc[1]:]
		} else {
			updated = original
		}
	} else {
		if replaceAll {
			count = strings.Count(original, find)
			updated = strings.ReplaceAll(original, find, replace)
		} else if idx := strings.Index(original, find); idx >= 0 {
			count = 1
			updated = original[:idx] + replace + original[idx+len(find):]
		} else {
			updated = original
		}
	}

	if count == 0 {
		return fmt.Sprintf("Pattern not found in %s — file unchanged.", cleanPath), nil
	}

	if err := os.WriteFile(cleanPath, []byte(updated), 0644); err != nil {
		return "", fmt.Errorf("failed to write file: %w", err)
	}

	noun := "occurrence"
	if count != 1 {
		noun = "occurrences"
	}
	return fmt.Sprintf("Replaced %d %s of %q in %s.", count, noun, find, cleanPath), nil
}

func fetchAndConvert(url string) (string, error) {
	client := &http.Client{}
	httpReq, err := http.NewRequest("GET", url, nil)
	if err != nil {
		return "", fmt.Errorf("invalid URL: %w", err)
	}
	httpReq.Header.Set("User-Agent", "MCP-Stdio-Server/1.0")

	resp, err := client.Do(httpReq)
	if err != nil {
		return "", fmt.Errorf("fetch failed: %w", err)
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		return "", fmt.Errorf("HTTP error: %d %s", resp.StatusCode, resp.Status)
	}

	body, err := io.ReadAll(resp.Body)
	if err != nil {
		return "", fmt.Errorf("read body failed: %w", err)
	}

	return HTMLToMarkdown(body)
}

func HTMLToMarkdown(htmlb []byte) (string, error) {
	html := string(htmlb)
	if strings.TrimSpace(html) == "" {
		return "", nil
	}

	converter := html2md.NewConverter("", true, nil)
	md, err := converter.ConvertString(html)
	if err != nil {
		return "", fmt.Errorf("html to md: %w", err)
	}

	return strings.TrimSpace(md), nil
}

func min(a, b int) int { if a < b { return a }; return b }

## Summary of Changes

| Area | Before | After |
|------|--------|-------|
| **Broken pipe** | No protection; raw `fmt.Fprintln` | `sync.Mutex` on stdout + explicit error logging on EPIPE |
| **Parse errors** | Generic "Parse error" | Includes truncated raw bytes, byte count, and parse message in `Data` |
| **Tool validation** | Silent zero-value defaults | Checks `arguments` presence, reports exact missing fields with a hint JSON example |
| **Unknown tool** | `"not found"` string | Returns list of all available tools so the client can suggest valid ones |
| **Signal handling** | None | Catches SIGINT/SIGTERM for clean shutdown (prevents hung processes) |
| **Notifications** | Ignored silently | Explicitly skipped when `ID` is nil per JSON-RPC spec |

The biggest win for AI usability is the `Data` field in error responses — it gives structured hints that LLMs can follow on retry, which dramatically reduces back-and-forth debugging cycles.
